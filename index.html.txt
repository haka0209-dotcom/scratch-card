            const savedRemaining = localStorage.getItem(STORAGE_KEY_REMAINING);

        // Display prize item layout
        function displayPrizeUnderCanvas(prize) {

        // --- 4. CANVAS SCRATCH FOIL LOGIC ---
        function setupCanvas() {
            canvas = document.getElementById('scratch-canvas');
            ctx = canvas.getContext('2d', { willReadFrequently: true });

            // Event Listeners for Scratching
            canvas.addEventListener('mousedown', startScratch);
            canvas.addEventListener('touchstart', startScratch, { passive: false });

            window.addEventListener('mousemove', scratchMove);
            window.addEventListener('touchmove', scratchMove, { passive: false });

            window.addEventListener('mouseup', endScratch);
            window.addEventListener('touchend', endScratch);

            // Handle Resize
            window.addEventListener('resize', () => {
                if (!isRevealed && currentAssignedPrize) {
                    resetCanvasFoil();
                }
            });
        }

        function resetCanvasFoil() {
            if (!canvas) return;
            const rect = canvas.getBoundingClientRect();
            
            // Safety check: Avoid setting 0 dimensions
            if (rect.width === 0 || rect.height === 0) return;

            canvas.width = rect.width;
            canvas.height = rect.height;

            totalPixels = canvas.width * canvas.height;
            isRevealed = false;

            // Draw Metallic Foil
            const gradient = ctx.createLinearGradient(0, 0, canvas.width, canvas.height);
            gradient.addColorStop(0, '#D4AF37'); // Gold
            gradient.addColorStop(0.25, '#FFF8DC');
            gradient.addColorStop(0.5, '#AA771C');
            gradient.addColorStop(0.75, '#FFD700');
            gradient.addColorStop(1, '#B8860B');

            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Add Decorative Pattern Text on Foil
            ctx.fillStyle = 'rgba(0, 0, 0, 0.15)';
            ctx.font = 'bold 16px Noto Sans TC, sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            
            const stepY = 40;
            const stepX = 80;
            for (let y = 20; y < canvas.height; y += stepY) {
                for (let x = 40; x < canvas.width; x += stepX) {
                    ctx.fillText('刮開此處', x, y);
                }
            }
        }

        function getPosition(e) {
            const rect = canvas.getBoundingClientRect();
            let clientX = e.clientX;
            let clientY = e.clientY;

            if (e.touches && e.touches.length > 0) {
                clientX = e.touches[0].clientX;
                clientY = e.touches[0].clientY;
            }

            return {
                x: clientX - rect.left,
                y: clientY - rect.top
            };
        }

        function startScratch(e) {
            if (isRevealed) return;
            isScratching = true;
            scratchMove(e);
        }

        function scratchMove(e) {
            if (!isScratching || isRevealed) return;
            if (e.cancelable) e.preventDefault();

            const pos = getPosition(e);

            ctx.globalCompositeOperation = 'destination-out';
            ctx.beginPath();
            ctx.arc(pos.x, pos.y, 22, 0, Math.PI * 2);
            ctx.fill();

            checkScratchPercentage();
        }

        function endScratch() {
            isScratching = false;
        }

        // Calculate cleared percentage to trigger auto-reveal
        function checkScratchPercentage() {
            if (isRevealed || !canvas || canvas.width === 0 || canvas.height === 0) return;

            try {
                // Sample pixels for performance
                const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
                const pixels = imageData.data;
                let clearedCount = 0;

                // Check alpha channel in increments
                for (let i = 3; i < pixels.length; i += 32) {
                    if (pixels[i] === 0) {
                        clearedCount++;
                    }
                }

                const totalSampled = pixels.length / 32;
                if (totalSampled === 0) return;
                
                const percentage = (clearedCount / totalSampled) * 100;

                if (percentage > 45) { // 45% cleared -> Auto Reveal
                    autoRevealPrize();
                }
            } catch (err) {
                console.warn("Scratch check skipped due to canvas dimension delay.", err);
            }
        }

        function getPosition(e) {
            const rect = canvas.getBoundingClientRect();
            let clientX = e.clientX;
            let clientY = e.clientY;

            if (e.touches && e.touches.length > 0) {
                clientX = e.touches[0].clientX;
                clientY = e.touches[0].clientY;
            }

            return {
                x: clientX - rect.left,
                y: clientY - rect.top
            };
        }

        function startScratch(e) {
            if (isRevealed) return;
            isScratching = true;
            scratchMove(e);
        }

        function scratchMove(e) {
            if (!isScratching || isRevealed) return;
            if (e.cancelable) e.preventDefault();

            const pos = getPosition(e);

            ctx.globalCompositeOperation = 'destination-out';
            ctx.beginPath();
            ctx.arc(pos.x, pos.y, 22, 0, Math.PI * 2);
            ctx.fill();

            checkScratchPercentage();
        }

        function endScratch() {
            isScratching = false;
        }

        // Calculate cleared percentage to trigger auto-reveal
        function checkScratchPercentage() {
            if (isRevealed) return;

            // Sample pixels for performance
            const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
            const pixels = imageData.data;
            let clearedCount = 0;

            // Check alpha channel in increments
            for (let i = 3; i < pixels.length; i += 32) {
                if (pixels[i] === 0) {
                    clearedCount++;
                }
            }

            const totalSampled = pixels.length / 32;
            const percentage = (clearedCount / totalSampled) * 100;

            if (percentage > 45) { // 45% cleared -> Auto Reveal
                autoRevealPrize();
            }
        }

        function autoRevealPrize() {
            isRevealed = true;
            // Clear canvas completely
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Trigger celebration confetti
            confetti({
                particleCount: 100,
                spread: 70,
                origin: { y: 0.6 }
            });

            // Show Winner Modal
            setTimeout(() => {
                showModal();
            }, 400);
        }

        function revealCanvasSilently() {
            isRevealed = true;
            if (ctx && canvas) {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
            }
        }

    // --- 6. ORGANIZER SECRET LIST LOGIC ---
    function handleSecretClick() {
        secretClickCount++;
        clearTimeout(secretClickTimer);

        if (secretClickCount >= 3) {
            secretClickCount = 0;
            openOrganizerModal();
        } else {
            secretClickTimer = setTimeout(() => {
                secretClickCount = 0;
            }, 1000);
        }
    }

    function openOrganizerModal() {
        const records = JSON.parse(localStorage.getItem(STORAGE_KEY_RECORDS) || "[]");
        const container = document.getElementById('organizer-records-list');

        if (records.length === 0) {
            container.innerHTML = `
                <div class="text-center py-8 text-gray-500">
                    <i class="fa-solid fa-inbox text-3xl mb-2"></i>
                    <p>目前尚無任何刮獎紀錄</p>
                </div>
            `;
        } else {
            container.innerHTML = records.map((rec, index) => {
                const date = new Date(rec.claimedAt).toLocaleString('zh-TW', {
                    month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit'
                });
                return `
                    <div class="p-3 bg-gray-800/80 rounded-xl border border-gray-700/60 flex items-center justify-between">
                        <div>
                            <div class="font-bold text-yellow-300 text-sm">#${index + 1} ${rec.userName}</div>
                            <div class="text-gray-300 font-medium text-xs mt-0.5">${rec.prize.name}</div>
                        </div>
                        <div class="text-[10px] text-gray-400 ml-2 whitespace-nowrap">${date}</div>
                    </div>
                `;
            }).join('');
        }

        const modal = document.getElementById('organizer-modal');
        const modalCard = document.getElementById('organizer-card');
        modal.classList.remove('opacity-0', 'pointer-events-none');
        modalCard.classList.remove('scale-90');
        modalCard.classList.add('scale-100');
    }

    function copyRecordsToClipboard() {
        const records = JSON.parse(localStorage.getItem(STORAGE_KEY_RECORDS) || "[]");
        if (records.length === 0) {
            alert('尚無紀錄可複製');
            return;
        }

        let text = "🏆 刮刮樂得獎名單紀錄 🏆\n---------------------\n";
        records.forEach((rec, i) => {
            text += `${i + 1}. 【${rec.userName}】 - ${rec.prize.name}\n`;
        });

        // Use standard copy trick
        const textarea = document.createElement('textarea');
        textarea.value = text;
        document.body.appendChild(textarea);
        textarea.select();
        document.execCommand('copy');
        document.body.removeChild(textarea);

        alert('得獎名單已成功複製到剪貼簿！');
    }
